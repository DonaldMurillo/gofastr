package router

import (
	"net/http"
	"net/http/httptest"
	"testing"
)

func TestReplayTwinServeHook(t *testing.T) {
	defer func() {
		if recover() != nil {
			t.Error("the serve hook's panic crossed the request")
		}
	}()
	r := New()
	r.SetServeHook(func(string, string) { panic("hook boom") })
	r.Get("/ok", http.HandlerFunc(func(w http.ResponseWriter, _ *http.Request) {
		w.WriteHeader(http.StatusTeapot)
	}))
	rec := httptest.NewRecorder()
	r.ServeHTTP(rec, httptest.NewRequest(http.MethodGet, "/ok", nil))
	if rec.Code != http.StatusTeapot {
		t.Errorf("status = %d, want the handler's 418", rec.Code)
	}
}
