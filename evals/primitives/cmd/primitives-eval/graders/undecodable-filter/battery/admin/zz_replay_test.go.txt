package admin

import (
	"strings"
	"testing"
	"time"

	"github.com/DonaldMurillo/gofastr/framework/entity"
)

const replayWarning = "Filter not applied"

func TestReplayReportedAudit(t *testing.T) {
	x := setup(t, map[string]entity.EntityConfig{"posts": postsConfig(), "notes": notesConfig()}, Config{Entities: []string{"posts", "notes"}}, nil)
	seedAuditDays(x, time.Now().UTC())
	if body := get(x.as(theAdmin), "/admin/audit?entity=posts%ZZ").Body.String(); !strings.Contains(body, replayWarning) {
		t.Error("an undecodable entity filter on the audit page drew no warning")
	}
	if body := get(x.as(theAdmin), "/admin/audit?entity=posts").Body.String(); strings.Contains(body, replayWarning) {
		t.Error("a decodable entity filter warned")
	}
}
