package uinoderender

import (
	"context"
	"testing"

	"github.com/DonaldMurillo/gofastr/core-ui/uinodev1"
)

func TestReplayTwinRendererDoesNotPanic(t *testing.T) {
	for _, cols := range [][]uinodev1.DataColumn{
		{{Key: "a", Label: "First"}, {Key: "a", Label: "Second"}},
		{{Key: "", Label: "Blank"}},
	} {
		func() {
			defer func() {
				if recover() != nil {
					t.Errorf("columns %v: the renderer panicked on a hand-built tree", cols)
				}
			}()
			r := New(context.Background(), nil)
			_, err := r.Render(tree(node(uinodev1.CompDataTable, uinodev1.DataTableProps{
				Columns: cols,
				Rows:    []uinodev1.DataRow{{Cells: []uinodev1.DataCell{{Text: "x"}, {Text: "y"}}}},
			})))
			if err == nil {
				t.Errorf("columns %v: want an error, got none", cols)
			}
		}()
	}
}
